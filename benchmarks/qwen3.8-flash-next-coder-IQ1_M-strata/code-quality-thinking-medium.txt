def shlex_split(s):
    tokens = []
    current = []
    i = 0
    in_token = False

    while i < len(s):
        c = s[i]

        # --- whitespace: end current token ---
        if c in ' \t\n\r':
            if in_token:
                tokens.append(''.join(current))
                current = []
                in_token = False
            i += 1
            continue

        # --- single-quoted run: everything literal, no escapes ---
        if c == "'":
            in_token = True
            i += 1
            while i < len(s):
                if s[i] == "'":
                    i += 1
                    break
                current.append(s[i])
                i += 1
            continue

        # --- double-quoted run ---
        if c == '"':
            in_token = True
            i += 1
            while i < len(s):
                if s[i] == '"':
                    i += 1
                    break
                if s[i] == '\\':
                    if i + 1 < len(s):
                        nxt = s[i + 1]
                        if nxt in ('\\', '"', '$', '`'):
                            current.append(nxt)
                            i += 2
                            continue
                        else:
                            current.append('\\')
                            i += 1
                            continue
                    else:
                        current.append('\\')
                        i += 1
                        continue
                current.append(s[i])
                i += 1
            continue

        # --- backslash outside quotes: escapes next char ---
        if c == '\\':
            in_token = True
            if i + 1 < len(s):
                current.append(s[i + 1])
                i += 2
            else:
                i += 1
            continue

        # --- plain character ---
        in_token = True
        current.append(c)
        i += 1

    if in_token:
        tokens.append(''.join(current))

    return tokens

def wrap_(text, width):
    # Preprocessing: expand tabs, replace/collapse all whitespace, strip
    text = text.expandtabs()
    text = ' '.join(text.split())

    if not text or width <= 0:
        return []

    # Split into words, then split each word on hyphens (hyphen stays with left part)
    chunks = []
    for word in text.split(' '):
        current = ''
        for c in word:
            current += c
            if c == '-':
                chunks.append(current)
                current = ''
        if current:
            chunks.append(current)

    if not chunks:
        return []

    # Greedy fill
    lines = []
    curr_line = []
    curr_len = 0

    for chunk in chunks:
        if len(chunk) > width:
            pieces = [chunk[i:i + width] for i in range(0, len(chunk), width)]
            for j, piece in enumerate(pieces):
                space_len = 1 if (curr_line and j == 0) else 0
                if curr_len + space_len + len(piece) <= width:
                    curr_line.append(piece)
                    curr_len += space_len + len(piece)
                else:
                    if curr_line:
                        lines.append(' '.join(curr_line))
                    curr_line = [piece]
                    curr_len = len(piece)
        else:
            space_len = 1 if curr_line else 0
            if curr_len + space_len + len(chunk) <= width:
                curr_line.append(chunk)
                curr_len += space_len + len(chunk)
            else:
                if curr_line:
                    lines.append(' '.join(curr_line))
                curr_line = [chunk]
                curr_len = len(chunk)

    if curr_line:
        lines.append(' '.join(curr_line))

    return lines

def add_months(y, m, d, n):
    def is_leap(year):
        return (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0)

    def days_in_month(year, month):
        if month == 2:
            return 29 if is_leap(year) else 28
        if month in (1, 3, 5, 7, 8, 10, 12):
            return 31
        return 30

    total = (m - 1) + n
    new_y = y + total // 12
    new_m = total % 12 + 1
    new_d = min(d, days_in_month(new_y, new_m))
    return (new_y, new_m, new_d)

def parse_qsl_(qs):
    def _unquote_plus(s):
        result = []
        i = 0
        n = len(s)
        while i < n:
            if s[i] == '+':
                result.append(' ')
                i += 1
            elif s[i] == '%' and i + 2 < n:
                hex_pair = s[i+1:i+3]
                try:
                    result.append(chr(int(hex_pair, 16)))
                    i += 3
                except ValueError:
                    result.append('%')
                    i += 1
            else:
                result.append(s[i])
                i += 1
        return ''.join(result)

    pairs = []
    for part in qs.split('&'):
        if not part:
            continue
        key, _, value = part.partition('=')
        pairs.append((_unquote_plus(key), _unquote_plus(value)))
    return pairs
