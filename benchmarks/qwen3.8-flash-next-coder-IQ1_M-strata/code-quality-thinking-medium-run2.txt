def glob_match(pattern, name):
    def match(pi, si):
        while pi < len(pattern):
            c = pattern[pi]
            if c == '*':
                while pi < len(pattern) and pattern[pi] == '*':
                    pi += 1
                if pi == len(pattern):
                    return True
                for si_try in range(si, len(name) + 1):
                    if match(pi, si_try):
                        return True
                return False
            elif c == '?':
                if si >= len(name):
                    return False
                pi += 1
                si += 1
            elif c == '[':
                j = pi + 1
                negated = False
                if j < len(pattern) and pattern[j] == '!':
                    negated = True
                    j += 1
                if j < len(pattern) and pattern[j] == ']':
                    j += 1
                while j < len(pattern) and pattern[j] != ']':
                    j += 1
                if j >= len(pattern):
                    if si >= len(name) or name[si] != '[':
                        return False
                    pi += 1
                    si += 1
                else:
                    if si >= len(name):
                        return False
                    ch = name[si]
                    in_class = False
                    k = pi + 1
                    if negated:
                        k += 1
                    while k < j:
                        if k + 2 < j and pattern[k + 1] == '-':
                            lo = pattern[k]
                            hi = pattern[k + 2]
                            if lo <= ch <= hi:
                                in_class = True
                                break
                            k += 3
                        else:
                            if pattern[k] == ch:
                                in_class = True
                                break
                            k += 1
                    if negated:
                        in_class = not in_class
                    if not in_class:
                        return False
                    pi = j + 1
                    si += 1
            else:
                if si >= len(name) or name[si] != c:
                    return False

        return si == len(name)

    return match(0, 0)

def urljoin_(base, url):
    def parse(s):
        frag = None
        if '#' in s:
            s, frag = s.split('#', 1)
        query = None
        if '?' in s:
            s, query = s.split('?', 1)
        scheme = None
        if s and s[0].isalpha():
            j = 1
            while j < len(s) and (s[j].isalnum() or s[j] in '.-'):
                j += 1
            if j < len(s) and s[j] == ':':
                scheme = s[:j]
                s = s[j + 1:]
        authority = None
        if s.startswith('//'):
            s = s[2:]
            idx = s.find('/')
            if idx == -1:
                authority = s
                s = ''
            else:
                authority = s[:idx]
                s = s[idx:]
        return scheme, authority, s, query, frag

    def normalize_path(path):
        if not path:
            return path
        abs = path.startswith('/')
        if abs:
            path = path[1:]
        parts = path.split('/')
        trailing = parts[-1] in ('.', '..') if parts else False
        result = []
        for part in parts:
            if part == '.':
                continue
            elif part == '..':
                if result:
                    result.pop()
            else:
                result.append(part)
        normalized = '/'.join(result)
        if abs:
            normalized = '/' + normalized
        if trailing and normalized and not normalized.endswith('/'):
            normalized += '/'
        return normalized

    def build(scheme, authority, path, query, fragment):
        r = ''
        if scheme is not None:
            r += scheme + ':'
        if authority is not None:
            r += '//' + authority
        r += path
        if query is not None:
            r += '?' + query
        if fragment is not None:
            r += '#' + fragment
        return r

    bs, ba, bp, bq, bf = parse(base)
    rs, ra, rp, rq, rf = parse(url)

    # 1. Reference has a scheme → absolute URL
    if rs is not None:
        return build(rs, ra, normalize_path(rp), rq, rf)

    # 2. Reference has authority (protocol-relative //host/path)
    if ra is not None:
        return build(bs, ra, normalize_path(rp), rq, rf)

    # 3. Reference has absolute path
    if rp.startswith('/'):
        return build(bs, ba, normalize_path(rp), rq, rf)

    # 4. Empty reference → base minus fragment
    if rp == '' and rq is None and rf is None:
        return build(bs, ba, bp, bq, None)

    # 5. Query-only reference
    if rp == '' and rq is not None:
        return build(bs, ba, bp, rq, rf)

    # 6. Fragment-only reference
    if rp == '' and rq is None and rf is not None:
        return build(bs, ba, bp, bq, rf)

    # 7. Relative path: replace last component of base path
    if bp == '':
        prefix = '/' if ba is not None else ''
    else:
        last_slash = bp.rfind('/')
        prefix = bp[:last_slash + 1] if last_slash != -1 else ''

    new_path = normalize_path(prefix + rp)
    query = rq if rq is not None else bq
    frag = rf

    return build(bs, ba, new_path, query, frag)

def wrap_(text, width):
    text = text.expandtabs(8)
    words = text.split()
    if not words:
        return []

    # Split words at hyphens (break_on_hyphens=True)
    chunks = []
    for word in words:
        if '-' in word:
            parts = word.split('-')
            for i, part in enumerate(parts):
                if i < len(parts) - 1:
                    chunks.append(part + '-')
                elif part:
                    chunks.append(part)
        else:
            chunks.append(word)

    # Greedy line-filling
    lines = []
    current_line = []
    current_len = 0  # length of ' '.join(current_line)

    for chunk in chunks:
        space = 1 if current_line else 0
        if current_len + space + len(chunk) > width:
            if current_line:
                lines.append(' '.join(current_line))
                current_line = []
                current_len = 0

        # Break long words into pieces of exactly `width`
        while len(chunk) > width:
            lines.append(chunk[:width])
            chunk = chunk[width:]

        if chunk:
            current_line.append(chunk)
            current_len += (1 if len(current_line) > 1 else 0) + len(chunk)

    if current_line:
        lines.append(' '.join(current_line))

    return lines

def add_months(y, m, d, n):
    total = y * 12 + (m - 1) + n
    new_y = total // 12
    new_m = total % 12 + 1
    if new_m == 2:
        last = 29 if (new_y % 4 == 0 and (new_y % 100 != 0 or new_y % 400 == 0)) else 28
    elif new_m in (1, 3, 5, 7, 8, 10, 12):
        last = 31
    else:
        last = 30
    return (new_y, new_m, min(d, last))

def parse_qsl_(qs):
    def unquote_plus(s):
        s = s.replace('+', ' ')
        b = s.encode('utf-8')
        result = bytearray()
        i = 0
        while i < len(b):
            if b[i] == 0x25 and i + 2 < len(b):
                hex_chars = chr(b[i+1]) + chr(b[i+2])
                try:
                    result.append(int(hex_chars, 16))
                    i += 3
                    continue
                except ValueError:
                    pass
            result.append(b[i])
            i += 1
        return bytes(result).decode('utf-8', errors='replace')

    pairs = []
    if not qs:
        return pairs
    for part in qs.split('&'):
        if not part:
            continue
        if '=' in part:
            key, value = part.split('=', 1)
        else:
            key, value = part, ''
        pairs.append((unquote_plus(key), unquote_plus(value)))
    return pairs
