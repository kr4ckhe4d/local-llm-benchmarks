def glob_match(pattern, name):
    def parse_class(pat, start):
        # pat[start] == '['; returns (members, negate, end_index) or None if unterminated
        i = start + 1
        negate = False
        if i < len(pat) and pat[i] == '!':
            negate = True
            i += 1
        members = []
        first = True
        while i < len(pat):
            if pat[i] == ']' and not first:
                return (members, negate, i)
            first = False
            if i + 2 < len(pat) and pat[i + 1] == '-' and pat[i + 2] != ']':
                members.append((pat[i], pat[i + 2]))
                i += 3
            else:
                members.append(pat[i])
                i += 1
        return None

    def match(pi, si):
        if pi == len(pattern):
            return si == len(name)
        c = pattern[pi]
        if c == '*':
            while pi < len(pattern) and pattern[pi] == '*':
                pi += 1
            if pi == len(pattern):
                return True
            for k in range(len(name) - si + 1):
                if match(pi, si + k):
                    return True
            return False
        if c == '?':
            if si >= len(name):
                return False
            return match(pi + 1, si + 1)
        if c == '[':
            result = parse_class(pattern, pi)
            if result is None:
                if si < len(name) and c == name[si]:
                    return match(pi + 1, si + 1)
                return False
            members, negate, end_pos = result
            if si >= len(name):
                return False
            ch = name[si]
            in_class = False
            for m in members:
                if isinstance(m, tuple):
                    if m[0] <= ch <= m[1]:
                        in_class = True
                        break
                else:
                    if m == ch:
                        in_class = True
                        break
            if negate:
                in_class = not in_class
            if in_class:
                return match(end_pos + 1, si + 1)
            return False
        if si < len(name) and c == name[si]:
            return match(pi + 1, si + 1)
        return False

    return match(0, 0)

def urljoin_(base, url):
    """Reproduce urllib.parse.urljoin using RFC 3986 reference resolution."""

    def parse(u):
        """Return (scheme, authority, path, query, fragment).
        authority is None if absent; query/fragment are None if absent."""
        scheme = ''
        authority = None
        path = ''
        query = None
        fragment = None

        if '#' in u:
            u, fragment = u.rsplit('#', 1)
        if '?' in u:
            u, query = u.rsplit('?', 1)

        # Extract scheme: ALPHA *(ALPHA / DIGIT / '-' / '.' / '_')
        colon = u.find(':')
        if colon > 0 and u[0].isalpha() and all(
            c.isalnum() or c in '-._+' for c in u[:colon]
        ):
            scheme = u[:colon]
            u = u[colon + 1:]

        # Extract authority (introduced by '//')
        if u.startswith('//'):
            u = u[2:]
            slash = u.find('/')
            if slash == -1:
                authority = u
                u = ''
            else:
                authority = u[:slash]
                u = u[slash:]

        path = u
        return (scheme, authority, path, query, fragment)

    def unparse(scheme, authority, path, query, fragment):
        result = ''
        if scheme:
            result += scheme + ':'
        if authority is not None:
            result += '//' + authority
            # Path after authority must be empty or start with '/'
            if path and path[0] != '/':
                path = '/' + path
        result += path
        if query is not None:
            result += '?' + query
        if fragment is not None:
            result += '#' + fragment
        return result

    def normalize_path(path):
        """Remove '.' and '..' components per RFC 3986 §5.2."""
        leading = path.startswith('/')
        parts = path.split('/')
        stack = []
        for part in parts:
            if part == '..':
                if stack and not (len(stack) == 1 and stack[0] == ''):
                    stack.pop()
            elif part == '.':
                pass
            else:
                stack.append(part)
        norm = '/'.join(stack)
        if leading:
            norm = '/' + norm
        return norm

    def resolve(base_str, url_str):
        bs, ba, bp, bq, bf = parse(base_str)
        us, ua, up, uq, uf = parse(url_str)

        # 1. Absolute URL: url has its own scheme → use it as-is
        if us:
            return unparse(us, ua, up, uq, uf)

        # 2. Protocol-relative: url has '//' but no scheme
        if ua is not None:
            return unparse(bs, ua, up, uq, uf)

        # 3. Empty path in url
        if up == '':
            if uq is not None:
                # Query-only reference "?q"
                return unparse(bs, ba, bp, uq, uf)
            elif uf is not None:
                # Fragment-only reference "#f"
                return unparse(bs, ba, bp, bq, uf)
            else:
                # Empty string: return base without its fragment
                return unparse(bs, ba, bp, bq, '')

        # 4. Absolute path in url
        if up.startswith('/'):
            return unparse(bs, ba, normalize_path(up), uq, uf)

        # 5. Relative path: replace last component of base path
        last_slash = bp.rfind('/')
        if last_slash == -1:
            base_path_part = ''
        else:
            base_path_part = bp[:last_slash + 1]

        new_path = normalize_path(base_path_part + up)
        return unparse(bs, ba, new_path, uq, uf)

    return resolve(base, url)

def wrap_(text, width):
    import re
    if width <= 0:
        raise ValueError("invalid width %r (must be > 0)" % width)
    text = text.expandtabs(8)
    text = ''.join(' ' if c.isspace() else c for c in text)
    chunks = [m.group() for m in re.finditer(r'(?:\w+\S*?[-])|(?:\S+)|(?:\s+)', text)]
    lines = []
    i = 0
    while i < len(chunks):
        cur_line = []
        cur_len = 0
        while i < len(chunks):
            l = len(chunks[i])
            if cur_len + l <= width:
                cur_line.append(chunks[i])
                cur_len += l
                i += 1
            else:
                break
        if i < len(chunks) and len(chunks[i]) > width:
            chunk = chunks[i]
            chunks[i] = chunk[:width]
            chunks.insert(i + 1, chunk[width:])
        if cur_line and cur_line[-1].strip() == '':
            cur_line.pop()
        if cur_line:
            lines.append(''.join(cur_line))
    return lines

def add_months(y, m, d, n):
    total = (m - 1) + n
    ty = y + total // 12
    tm = total % 12 + 1

    def days_in_month(year, month):
        if month == 2:
            if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
                return 29
            return 28
        if month in (1, 3, 5, 7, 8, 10, 12):
            return 31
        return 30

    td = min(d, days_in_month(ty, tm))
    return (ty, tm, td)
