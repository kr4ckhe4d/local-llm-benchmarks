                #       - But we need to handle the case where the line starts with whitespace.
                #       - Example: current_line = [('ws','  '), ('word','a')] -> "  a"
                #       - So we just join the tokens in order.

                #   However, we need to be careful: 
                #       - When we break a line, we pop the trailing whitespace, so the current_line might end with a word.
                #       - Then we add the next token (which might be whitespace or word) to the new line.
                #       - So the conversion is straightforward: join the tokens.

                #   But what if the line is empty? 
                #       - Example: "  a" with width=2 -> 
                #           Tokens: [('ws','  '), ('word','a')]
                #           Line 1: add '  ' -> current_line = [('ws','  ')]
                #           Next: 'a'. Try to add: len=2+1=3>2 -> break.
                #           Pop trailing: current_line.pop() -> current_line = []
                #           Start new line with 'a'.
                #           current_line = [('word','a')]
                #           End: lines = ["a"]
                #       - So the first line was popped, so it's empty.
                #       - So we only add non-empty lines.

                #   So after the loop, if current_line is not empty, we add it to lines.
                #   But we need to convert current_line to a string.
                #   How? 
                #       - We can join the tokens: 
                #           line_str = "".join(token[1] for token in current_line)
                #       - But we need to make sure that the whitespace tokens are included.
                #       - Example: [('ws','  '), ('word','a')] -> "  a"
                #       - So yes, just join.

                #   But what if the line starts with whitespace? 
                #       - Example: "  a" with width=5 -> 
                #           Tokens: [('ws','  '), ('word','a')]
                #           Line 1: add '  ' -> len=2.
                #           Next: 'a' -> len=2+1=3<=5 -> add.
                #           current_line = [('ws','  '), ('word','a')]
                #           line_str = "  a"
                #       - Correct.

                #   So the conversion is: 
                #       line_str = "".join(token[1] for token in current_line)

                #   But we need to handle the case where the line is empty.
                #       - If current_line is empty, skip.

                #   So:
                #       if current_line:
                #           line_str = "".join(token[1] for token in current_line)
                #           lines.append(line_str)

                #   But wait, what if the line has only whitespace? 
                #       - Example: "  " with width=2 -> 
                #           Tokens: [('ws','  ')]
                #           Line 1: add '  ' -> current_line = [('ws','  ')]
                #           End: line_str = "  "
                #           lines = ["  "]
                #       - Correct.

                #   So the algorithm is:
                #       1. Tokenize the text.
                #       2. Break long words into chunks.
                #       3. Process tokens to build lines.
                #       4. Convert each line to a string.

                #   Let's implement.

    # Preprocess: break long words
    def break_long_words(tokens, w):
        new_tokens = []
        for token_type, token in tokens:
            if token_type == 'word' and len(token) > w:
                chunks = []
                while token:
                    if len(token) <= w:
                        chunks.append(token)
                        break
                    else:
                        chunks.append(token[:w])
                        token = token[w:]
                new_tokens.extend([('word', chunk) for chunk in chunks])
            else:
                new_tokens.append((token_type, token))
        return new_tokens

    # Process tokens to build lines
    current_line = []
    current_length = 0
    on_first_line = True
    lines = []

    def get_line_length():
        total = 0
        for token_type, token in current_line:
            total += len(token)
        return total

    for token_type, token in break_long_words(tokens, width):
        if token_type == 'ws':
            if current_line:
                # Try to add
                if current_length + len(token) <= width:
                    current_line.append((token_type, token))
                    current_length += len(token)
                else:
                    # Break the line
                    # Pop trailing whitespace
                    if current_line and current_line[-1][0] == 'ws':
                        current_line.pop()
                        current_length = get_line_length()
                    # Start a new line with this whitespace token
                    if on_first_line:
                        current_line.append((token_type, token))
                        current_length = len(token)
                    else:
                        # Leading whitespace of new line -> drop it
                        current_line = []
                        current_length = 0
                    on_first_line = False
            else:
                # current_line is empty
                if on_first_line:
                    current_line.append((token_type, token))
                    current_length = len(token)
                else:
                    # Leading whitespace of new line -> drop it
                    current_line = []
                    current_length = 0
        else: # 'word'
            if current_line:
                # Try to add
                if current_length + len(token) <= width:
                    current_line.append((token_type, token))
                    current_length += len(token)
                else:
                    # Break the line
                    # Pop trailing whitespace
                    if current_line and current_line[-1][0] == 'ws':
                        current_line.pop()
                        current_length = get_line_length()
                    # Start a new line with this word
                    current_line.append((token_type, token))
                    current_length = len(token)
                    on_first_line = False
            else:
                current_line.append((token_type, token))
                current_length = len(token)
                on_first_line = False

    # Add the last line if not empty
    if current_line:
        line_str = "".join(token[1] for token in current_line)
        lines.append(line_str)

    return lines
