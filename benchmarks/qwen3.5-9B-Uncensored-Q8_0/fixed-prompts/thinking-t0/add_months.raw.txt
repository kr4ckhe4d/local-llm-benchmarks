```python
import calendar

def add_months(y, m, d, n):
    # Calculate total months from a reference point (Year 0, Month 1)
    # Using (m - 1) to make it 0-indexed for easier modulo arithmetic
    total_months = y * 12 + (m - 1) + n
    
    # Calculate new year and month
    new_y = total_months // 12
    new_m = (total_months % 12) + 1
    
    # Determine the number of days in the target month
    # calendar.monthrange returns (weekday_of_first_day, number_of_days)
    _, days_in_target_month = calendar.monthrange(new_y, new_m)
    
    # Clamp the day if it exceeds the days in the target month
    if d > days_in_target_month:
        d = days_in_target_month
    
    return (new_y, new_m, d)
```
