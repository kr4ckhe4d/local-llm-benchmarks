def shlex_split(s):
    """
    Reproduce shlex.split(s) in POSIX mode from scratch.
    """
    if not s:
        return []

    tokens = []
    current_token = []
    i = 0
    n = len(s)
    
    # State: 0 = unquoted, 1 = single quoted, 2 = double quoted
    state = 0
    
    while i < n:
        char = s[i]
        
        if state == 0:
            # Unquoted state
            if char == ' ':
                if current_token:
                    tokens.append(''.join(current_token))
                    current_token = []
                i += 1
                continue
            elif char == "'":
                # Start single quoted
                state = 1
                i += 1
                continue
            elif char == '"':
                # Start double quoted
                state = 2
                i += 1
                continue
            elif char == '\\':
                # Escape next char
                if i + 1 < n:
                    next_char = s[i+1]
                    current_token.append(next_char)
                    i += 2
                    continue
                else:
                    # Trailing backslash, treat as literal
                    current_token.append(char)
                    i += 1
                    continue
            else:
                current_token.append(char)
                i += 1
                continue
        
        elif state == 1:
            # Single quoted state
            if char == "'":
                # End single quoted
                state = 0
                i += 1
                continue
            else:
                # Everything literal inside
                current_token.append(char)
                i += 1
                continue
        
        elif state == 2:
            # Double quoted state
            if char == '"':
                # End double quoted
                state = 0
                i += 1
                continue
            elif char == '\\':
                # Backslash escapes only another backslash or a double quote
                if i + 1 < n:
                    next_char = s[i+1]
                    if next_char == '\\' or next_char == '"':
                        current_token.append(next_char)
                        i += 2
                        continue
                    else:
                        # Other backslashes are literal
                        current_token.append(char)
                        i += 1
                        continue
                else:
                    current_token.append(char)
                    i += 1
                    continue
            else:
                current_token.append(char)
                i += 1
                continue
    
    # Append remaining token if any
    if current_token:
        tokens.append(''.join(current_token))
    
    return tokens
