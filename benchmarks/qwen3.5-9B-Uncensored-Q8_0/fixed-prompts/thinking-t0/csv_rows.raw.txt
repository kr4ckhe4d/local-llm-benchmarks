def csv_rows(text):
    rows = []
    current_row = []
    current_field = []
    in_quotes = False
    i = 0
    n = len(text)

    while i < n:
        char = text[i]

        if in_quotes:
            if char == '"':
                # Check for escaped quote ("")
                if i + 1 < n and text[i+1] == '"':
                    current_field.append('"')
                    i += 2
                    continue
                else:
                    # End of quoted field
                    in_quotes = False
                    i += 1
                    continue
            elif char == '\n':
                # End of row (LF)
                current_row.append(''.join(current_field))
                rows.append(current_row)
                current_row = []
                current_field = []
                i += 1
                continue
            elif char == '\r':
                # Check for CRLF
                if i + 1 < n and text[i+1] == '\n':
                    current_row.append(''.join(current_field))
                    rows.append(current_row)
                    current_row = []
                    current_field = []
                    i += 2
                    continue
                else:
                    # Just CR inside quotes
                    current_row.append(''.join(current_field))
                    rows.append(current_row)
                    current_row = []
                    current_field = []
                    i += 1
                    continue
            else:
                current_field.append(char)
                i += 1
                continue
        else:
            # Outside quotes
            if char == '"':
                in_quotes = True
                i += 1
                continue
            elif char == ',':
                current_row.append(''.join(current_field))
                current_field = []
                i += 1
                continue
            elif char == '\n':
                current_row.append(''.join(current_field))
                rows.append(current_row)
                current_row = []
                current_field = []
                i += 1
                continue
            elif char == '\r':
                # Check for CRLF
                if i + 1 < n and text[i+1] == '\n':
                    current_row.append(''.join(current_field))
                    rows.append(current_row)
                    current_row = []
                    current_field = []
                    i += 2
                    continue
                else:
                    # Just CR outside quotes
                    current_row.append(''.join(current_field))
                    rows.append(current_row)
                    current_row = []
                    current_field = []
                    i += 1
                    continue
            else:
                current_field.append(char)
                i += 1
                continue

    # Handle the last field/row if the file doesn't end with a newline
    if current_field or current_row:
        current_row.append(''.join(current_field))
        rows.append(current_row)

    return rows
