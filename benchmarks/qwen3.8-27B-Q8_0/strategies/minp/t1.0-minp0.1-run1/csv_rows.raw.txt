```python
def csv_rows(text):
    """
    Parses RFC 4180 CSV text and returns a list of rows,
    each a list of string fields.
    """
    rows = []
    row = []
    field = []
    in_quotes = False
    i = 0
    n = len(text)
    
    while i < n:
        c = text[i]
        
        if in_quotes:
            if c == '"':
                # Check for doubled quote (escaped quote)
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 2
                else:
                    # End of quoted field
                    in_quotes = False
                    i += 1
            else:
                field.append(c)
                i += 1
        else:
            if c == '"':
                # Start of quoted field
                in_quotes = True
                i += 1
            elif c == ',':
                # End of field
                row.append(''.join(field))
                field = []
                i += 1
            elif c == '\r':
                # Handle CRLF or standalone CR
                if i + 1 < n and text[i + 1] == '\n':
                    # CRLF
                    row.append(''.join(field))
                    rows.append(row)
                    row = []
                    field = []
                    i += 2
                else:
                    # Standalone CR
                    row.append(''.join(field))
                    rows.append(row)
                    row = []
                    field = []
                    i += 1
            elif c == '\n':
                # LF
                row.append(''.join(field))
                rows.append(row)
                row = []
                field = []
                i += 1
            else:
                field.append(c)
                i += 1
    
    # After processing all characters, if there's remaining field content
    # or we're at the end, append the last field and row if not empty
    # But we need to be careful: if text ends with a newline, the last
    # row was already appended. If not, we need to append the remaining row.
    
    # Check if we have any remaining fields or if the last character wasn't a newline
    # We'll append the last row if field has content or if row has fields
    # Actually, we should append the last row if we have any pending data
    if field or row:
        row.append(''.join(field))
        rows.append(row)
    
    return rows
```
