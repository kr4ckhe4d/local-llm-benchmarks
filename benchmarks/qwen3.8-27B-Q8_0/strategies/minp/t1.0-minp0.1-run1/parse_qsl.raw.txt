def parse_qsl_(qs):
    """
    Parse a query string into a list of (key, value) pairs.
    
    - Separator is only '&'
    - '+' is decoded as space
    - Percent-escapes are decoded
    - Blank values are kept
    - Components without '=' yield (key, '')
    """
    if not qs:
        return []
    
    result = []
    pairs = qs.split('&')
    
    for pair in pairs:
        if not pair:
            continue
        
        # Split on first '='
        if '=' in pair:
            key, value = pair.split('=', 1)
        else:
            key, value = pair, ''
        
        # Decode percent-escapes and '+' as space
        key = _unquote(key)
        value = _unquote(value)
        
        result.append((key, value))
    
    return result


def _unquote(s):
    """Decode percent-escapes and '+' as space."""
    # Replace '+' with space
    s = s.replace('+', ' ')
    
    # Decode percent-escapes
    result = []
    i = 0
    while i < len(s):
        if s[i] == '%' and i + 2 < len(s):
            hex_chars = s[i+1:i+3]
            try:
                byte_val = int(hex_chars, 16)
                result.append(chr(byte_val))
                i += 3
            except (ValueError, TypeError):
                result.append(s[i])
                i += 1
        else:
            result.append(s[i])
            i += 1
    
    return ''.join(result)
