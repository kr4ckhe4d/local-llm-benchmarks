def glob_match(pattern, name):
    # Preprocess: parse the pattern into a list of tokens
    # Each token is either:
    #   ('star',)
    #   ('question',)
    #   ('char', c)
    #   ('class', negated, chars_set)  where chars_set is a set of characters
    #   ('literal', c)
    
    tokens = []
    i = 0
    n = len(pattern)
    while i < n:
        c = pattern[i]
        if c == '*':
            tokens.append(('star',))
            i += 1
        elif c == '?':
            tokens.append(('question',))
            i += 1
        elif c == '[':
            # Try to parse a character class
            # Check for negation
            j = i + 1
            negated = False
            if j < n and pattern[j] == '!':
                negated = True
                j += 1
            # A ']' immediately after '[' or '[!' is a literal
            if j < n and pattern[j] == ']':
                # literal ']'
                # The '[' is treated as literal '['
                tokens.append(('char', '['))
                i += 1
                continue
            # Parse the class contents
            chars = set()
            end = -1
            j += 1  # move past possible '!'
            # We need to find the closing ']'
            # But we also need to handle ranges
            k = j
            found_close = False
            while k < n:
                if pattern[k] == ']':
                    end = k
                    found_close = True
                    break
                k += 1
            if not found_close:
                # Unterminated '[', treat as literal '['
                tokens.append(('char', '['))
                i += 1
                continue
            # Now parse from j to end-1 for the class contents
            # Handle ranges like a-z
            m = j
            valid = True
            while m < end:
                ch = pattern[m]
                if m + 2 < end and pattern[m+1] == '-' and pattern[m+2] != ']':
                    # Range
                    start_ch = ch
                    end_ch = pattern[m+2]
                    if start_ch <= end_ch:
                        for code in range(ord(start_ch), ord(end_ch) + 1):
                            chars.add(chr(code))
                        m += 3
                    else:
                        # Invalid range, treat as literal? Or just skip?
                        # Let's treat the characters as literals
                        chars.add(ch)
                        m += 1
                else:
                    chars.add(ch)
                    m += 1
            tokens.append(('class', negated, chars))
            i = end + 1
        else:
            tokens.append(('char', c))
            i += 1
    
    # Now do matching with backtracking for '*'
    # Use recursive matching with memoization or simple recursion
    # Since patterns can have multiple '*', we need backtracking
    
    def match(pos, tpos):
        # pos: current position in name
        # tpos: current position in tokens
        if tpos == len(tokens):
            return pos == len(name)
        
        token = tokens[tpos]
        
        if token[0] == 'star':
            # '*' matches any sequence of characters
            # Try matching the rest of the pattern at each position from pos to len(name)
            for k in range(pos, len(name) + 1):
                if match(k, tpos + 1):
                    return True
            return False
        elif token[0] == 'question':
            if pos >= len(name):
                return False
            return match(pos + 1, tpos + 1)
        elif token[0] == 'char':
            if pos >= len(name):
                return False
            if name[pos] != token[1]:
                return False
            return match(pos + 1, tpos + 1)
        elif token[0] == 'class':
            if pos >= len(name):
                return False
            negated = token[1]
            char_set = token[2]
            ch = name[pos]
            if negated:
                if ch not in char_set:
                    return match(pos + 1, tpos + 1)
                else:
                    return False
            else:
                if ch in char_set:
                    return match(pos + 1, tpos + 1)
                else:
                    return False
        return False
    
    return match(0, 0)
