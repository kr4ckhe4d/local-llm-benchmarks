def csv_rows(text):
    """Parse RFC 4180 CSV text and return a list of rows, each a list of string fields."""
    rows = []
    row = []
    field = []
    i = 0
    n = len(text)
    
    in_quotes = False
    field_started = False  # whether we've started reading a field (to handle empty fields)
    
    while i < n:
        c = text[i]
        
        if in_quotes:
            if c == '"':
                # Check if next char is also a quote (escaped quote)
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 2
                    continue
                else:
                    # End of quoted field
                    in_quotes = False
                    i += 1
                    continue
            else:
                field.append(c)
                i += 1
                continue
        else:
            if not field_started and c == '"':
                # Start of a quoted field
                in_quotes = True
                field_started = True
                i += 1
                continue
            elif c == ',':
                # End of field, start new one
                row.append(''.join(field))
                field = []
                field_started = False
                i += 1
                continue
            elif c == '\r':
                # Handle CRLF
                if i + 1 < n and text[i + 1] == '\n':
                    row.append(''.join(field))
                    field = []
                    field_started = False
                    rows.append(row)
                    row = []
                    i += 2
                    continue
                else:
                    # Lone CR, treat as line ending
                    row.append(''.join(field))
                    field = []
                    field_started = False
                    rows.append(row)
                    row = []
                    i += 1
                    continue
            elif c == '\n':
                # LF line ending
                row.append(''.join(field))
                field = []
                field_started = False
                rows.append(row)
                row = []
                i += 1
                continue
            else:
                field.append(c)
                field_started = True
                i += 1
                continue
    
    # Handle the last field/row if there's remaining content
    if field or row or field_started:
        row.append(''.join(field))
        rows.append(row)
    
    return rows
