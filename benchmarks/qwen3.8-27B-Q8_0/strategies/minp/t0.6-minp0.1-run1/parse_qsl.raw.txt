def parse_qsl_(qs):
    if not qs:
        return []
    
    pairs = []
    # Split on '&' only
    components = qs.split('&')
    
    for component in components:
        if not component:
            continue
        
        # Split on first '=' only
        if '=' in component:
            key, value = component.split('=', 1)
        else:
            key = component
            value = ''
        
        # Decode key and value: replace '+' with space, then decode percent-escapes
        key = _decode_component(key)
        value = _decode_component(value)
        
        pairs.append((key, value))
    
    return pairs


def _decode_component(s):
    # First, replace '+' with space
    s = s.replace('+', ' ')
    
    # Decode percent-escapes
    result = []
    i = 0
    while i < len(s):
        if s[i] == '%' and i + 2 < len(s) + 1 and i + 2 < len(s):
            hex_str = s[i+1:i+3]
            try:
                byte_val = int(hex_str, 16)
                result.append(chr(byte_val))
                i += 3
            except ValueError:
                result.append(s[i])
                i += 1
        else:
            result.append(s[i])
            i += 1
    
    return ''.join(result)
