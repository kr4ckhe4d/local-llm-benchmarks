def parse_qsl_(qs):
    """
    Parse a query string into a list of (key, value) pairs.
    
    - Splits on '&' only.
    - Decodes percent-escapes and '+' as space.
    - Keeps blank values.
    - A component with no '=' yields a pair with an empty value.
    """
    def unquote(s):
        """Decode percent-escapes and '+' as space."""
        result = []
        i = 0
        n = len(s)
        while i < n:
            c = s[i]
            if c == '+':
                result.append(' ')
                i += 1
            elif c == '%':
                if i + 2 < n:
                    hex_str = s[i+1:i+3]
                    try:
                        code = int(hex_str, 16)
                        result.append(chr(code))
                        i += 3
                    except ValueError:
                        result.append(c)
                        i += 1
                else:
                    result.append(c)
                    i += 1
            else:
                result.append(c)
                i += 1
        return ''.join(result)
    
    if not qs:
        return []
    
    pairs = []
    for component in qs.split('&'):
        if not component:
            continue
        if '=' in component:
            key, value = component.split('=', 1)
        else:
            key = component
            value = ''
        pairs.append((unquote(key), unquote(value)))
    
    return pairs
