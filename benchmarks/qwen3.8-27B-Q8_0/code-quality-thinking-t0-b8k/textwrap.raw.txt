def wrap_(text, width):
    if width < 1:
        raise ValueError("width must be > 0")

    # Expand tabs exactly like str.expandtabs(8).
    expanded = []
    col = 0
    for ch in text:
        if ch == "\t":
            n = 8 - (col % 8)
            expanded.append(" " * n)
            col += n
        else:
            expanded.append(ch)
            col += 1
    text = "".join(expanded)

    # Collapse all whitespace to single spaces.
    words = []
    cur = []
    for ch in text:
        if ch in " \t\n\r\f\v":
            if cur:
                words.append("".join(cur))
                cur = []
        else:
            cur.append(ch)
    if cur:
        words.append("".join(cur))

    lines = []
    line = ""

    for word in words:
        if not line:
            if len(word) <= width:
                line = word
            else:
                while word:
                    take = min(width, len(word))
                    lines.append(word[:take])
                    word = word[take:]
        else:
            candidate = line + " " + word
            if len(candidate) <= width:
                line = candidate
            else:
                lines.append(line)
                line = ""
                if len(word) <= width:
                    line = word
                else:
                    while word:
                        take = min(width, len(word))
                        lines.append(word[:take])
                        word = word[take:]

    if line:
        lines.append(line)

    return lines
