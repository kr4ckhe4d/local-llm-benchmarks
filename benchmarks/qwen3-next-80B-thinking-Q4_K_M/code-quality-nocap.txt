def glob_match(pattern, name):

        return not name
    if not name:
        return all(c == '*' for c in pattern)

        return glob_match(pattern[1:], name[1:])
    elif pattern[0] == '[':
        j = 1
        while j < len(pattern) and pattern[j] != ']':
            j += 1
        if j == len(pattern):
            if name and name[0] == '[':
                return glob_match(pattern[1:], name[1:])
            return False
        if j == 1:
            if name and name[0] == '[':

                i += 1

            return False

def wrap_(text, width):
    current_pos = 0
    expanded = []
    for char in text:
        if char == '\t':
            num_spaces = 8 - (current_pos % 8)
            expanded.append(' ' * num_spaces)
            current_pos += num_spaces
        else:
            expanded.append(char)
            current_pos += 1
    expanded = ''.join(expanded)
    collapsed = ' '.join(expanded.split())
    words = collapsed.split()
    lines = []
    current_line = ''
    for word in words:
        if len(word) > width:
            for i in range(0, len(word), width):
                chunk = word[i:i+width]
                if current_line == '':
                    current_line = chunk
                else:
                    if len(current_line) + 1 + len(chunk) <= width:
                        current_line += ' ' + chunk
                    else:
                        lines.append(current_line)
                        current_line = chunk
        else:

                current_line = word
            else:
                if len(current_line) + 1 + len(word) <= width:
                    current_line += ' ' + word
                else:
                    lines.append(current_line)
                    current_line = word
    if current_line != '':
        lines.append(current_line)

import calendar

def add_months(y, m, d, n):
    total = y * 12 + m - 1 + n
    new_y = total // 12
    new_m = total % 12 + 1
    _, last_day = calendar.monthrange(new_y, new_m)
    new_d = min(d, last_day)
    return (new_y, new_m, new_d)
