def shlex_split(s):
    tokens = []
    current_token = []
    i = 0
    n = len(s)
    
    while i < n:
        char = s[i]
        
        # Handle whitespace
        if char in ' \t\n\r':
            if current_token:
                tokens.append(''.join(current_token))
                current_token = []
            i += 1
            continue
        
        # Handle single quotes
        if char == "'":
            i += 1
            start = i
            while i < n and s[i] != "'":
                i += 1
            current_token.extend(s[start:i])
            if i < n:
                i += 1  # Skip closing quote
            continue
        
        # Handle double quotes
        if char == '"':
            i += 1
            start = i
            while i < n and s[i] != '"':
                if s[i] == '\\' and i + 1 < n:
                    # Inside double quotes, backslash escapes only \ or "
                    next_char = s[i+1]
                    if next_char in ('\\', '"'):
                        current_token.append(s[i])
                        current_token.append(next_char)
                        i += 2
                        continue
                    else:
                        # Backslash is literal for other chars
                        current_token.append(s[i])
                        i += 1
                else:
                    current_token.append(s[i])
                    i += 1
            current_token.extend(s[start:i])
            if i < n:
                i += 1  # Skip closing quote
            continue
        
        # Handle backslash escaping outside quotes
        if char == '\\' and i + 1 < n:
            current_token.append(char)
            current_token.append(s[i+1])
            i += 2
            continue
        
        # Regular character
        current_token.append(char)
        i += 1
    
    # Append remaining token
    if current_token:
        tokens.append(''.join(current_token))
    
    return tokens
