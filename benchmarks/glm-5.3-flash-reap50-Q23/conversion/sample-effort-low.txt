def fnmatch_simple(name, pattern):
    def parse_class(i, ch):
        j = i + 1
        negate = False
        if j < len(pattern) and pattern[j] == '!':
            negate = True
            j += 1
        matched = False
        while j < len(pattern) and pattern[j] != ']':
            if j + 2 < len(pattern) and pattern[j + 1] == '-' and pattern[j + 2] != ']':
                if pattern[j] <= ch <= pattern[j + 2]:
                    matched = True
                j += 3
            else:
                if pattern[j] == ch:
                    matched = True
                j += 1
        if j >= len(pattern):
            return False, j
        return matched != negate, j + 1

    def rec(ni, pi):
        while pi < len(pattern):
            c = pattern[pi]
            if c == '*':
                for k in range(ni, len(name) + 1):
                    if rec(k, pi + 1):
                        return True
                return False
            elif c == '?':
                if ni >= len(name):
                    return False
                ni += 1
                pi += 1
            elif c == '[':
                if ni >= len(name):
                    return False
                ok, npext = parse_class(pi, name[ni])
                if not ok:
                    return False
                ni += 1
                pi = npext
            else:
                if ni >= len(name) or name[ni] != c:
                    return False
                ni += 1
                pi += 1
        return ni == len(name)

    return rec(0, 0)
```

Note: parse_class takes i (index of '['), so j = i+1. Range check for '-' needs j+2 < len(pattern) — but if '-' is last char before end, treat literally. Also `pattern[j+1]=='-'` where j+2 could equal len → literal. Fine.

Edge: unterminated class like `[abc` — fnmatch treats it as literal? fnmatch returns False for unterminated. Our parse_class returns False → no match. Acceptable.
