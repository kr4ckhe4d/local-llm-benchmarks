def add_months(y, m, d, n):
    total = y * 12 + (m - 1) + n
    y2, m2 = divmod(total, 12)
    m2 += 1
    if m2 == 2:
        last = 29 if (y2 % 4 == 0 and (y2 % 100 != 0 or y2 % 400 == 0)) else 28
    elif m2 in (4, 6, 9, 11):
        last = 30
    else:
        last = 31
    return (y2, m2, min(d, last))
