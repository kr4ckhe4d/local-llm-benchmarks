def csv_rows(text):
    rows = []
    current_row = []
    current_field = []
    state = 0  # 0: normal, 1: inside quoted field, 2: after quote inside quoted field
    i = 0
    n = len(text)

    while i < n:
        c = text[i]

        if state == 0:
            if c == '"':
                state = 1
            elif c == ',':
                current_row.append(''.join(current_field))
                current_field = []
            elif c == '\r':
                current_row.append(''.join(current_field))
                current_field = []
                rows.append(current_row)
                current_row = []
                if i + 1 < n and text[i + 1] == '\n':
                    i += 1
            elif c == '\n':
                current_row.append(''.join(current_field))
                current_field = []
                rows.append(current_row)
                current_row = []
            else:
                current_field.append(c)

        elif state == 1:
            if c == '"':
                state = 2
            else:
                current_field.append(c)

        else:  # state == 2
            if c == '"':
                current_field.append('"')
                state = 1
            elif c == ',':
                current_row.append(''.join(current_field))
                current_field = []
                state = 0
            elif c == '\r':
                current_row.append(''.join(current_field))
                current_field = []
                rows.append(current_row)
                current_row = []
                if i + 1 < n and text[i + 1] == '\n':
                    i += 1
                state = 0
            elif c == '\n':
                current_row.append(''.join(current_field))
                current_field = []
                rows.append(current_row)
                current_row = []
                state = 0
            else:
                # Non-strict: treat as literal quote followed by ordinary character
                current_field.append('"')
                current_field.append(c)
                state = 1

        i += 1

    if current_row or current_field or state != 0:
        current_row.append(''.join(current_field))
        rows.append(current_row)

    return rows
