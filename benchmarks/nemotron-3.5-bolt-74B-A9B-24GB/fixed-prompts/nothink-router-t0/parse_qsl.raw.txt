def parse_qsl_(qs):
    pairs = []
    for component in qs.split('&'):
        if '=' in component:
            key, value = component.split('=', 1)
        else:
            key, value = component, ''
        key = key.replace('+', ' ')
        value = value.replace('+', ' ')
        key = __import__('urllib.parse').unquote(key)
        value = __import__('urllib.parse').unquote(value)
        pairs.append((key, value))
    return pairs
