def add_months(y, m, d, n):
    """
    Return a (year, month, day) tuple n calendar months after the given date.
    If the day does not exist in the target month, clamp to the last day of that month.
    n may be negative.
    """
    import calendar
    
    # Calculate new year and month
    total_months = (y * 12 + (m - 1)) + n
    new_y = total_months // 12
    new_m = total_months % 12 + 1
    
    # Clamp day to the last day of the new month
    last_day = calendar.monthrange(new_y, new_m)[1]
    new_d = min(d, last_day)
    
    return (new_y, new_m, new_d)
