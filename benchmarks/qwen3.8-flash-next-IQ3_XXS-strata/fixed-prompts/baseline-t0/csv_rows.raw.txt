def csv_rows(text):
    """
    Parse RFC 4180 CSV text and return a list of rows, each a list of string fields.
    Handles quoted fields with commas/newlines, doubled quotes as escaped quotes,
    and CRLF or LF line endings.
    """
    rows = []
    row = []
    field = []
    in_quotes = False
    i = 0
    n = len(text)

    while i < n:
        ch = text[i]

        if in_quotes:
            if ch == '"':
                # Check for doubled quote
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 2
                else:
                    # End of quoted field
                    in_quotes = False
                    i += 1
            else:
                field.append(ch)
                i += 1
        else:
            if ch == '"':
                # Start of quoted field
                in_quotes = True
                i += 1
            elif ch == ',':
                # End of field
                row.append(''.join(field))
                field = []
                i += 1
            elif ch == '\r':
                # Handle CRLF or lone CR
                if i + 1 < n and text[i + 1] == '\n':
                    # CRLF
                    row.append(''.join(field))
                    field = []
                    rows.append(row)
                    row = []
                    i += 2
                else:
                    # Lone CR (treat as line ending)
                    row.append(''.join(field))
                    field = []
                    rows.append(row)
                    row = []
                    i += 1
            elif ch == '\n':
                # LF line ending
                row.append(''.join(field))
                field = []
                rows.append(row)
                row = []
                i += 1
            else:
                field.append(ch)
                i += 1

    # Handle last field/row if text doesn't end with a newline
    if field or row:
        row.append(''.join(field))
        rows.append(row)

    return rows
