def parse_qsl_(qs):
    def unquote(s):
        res = []
        i = 0
        while i < len(s):
            if s[i] == '+':
                res.append(' ')
                i += 1
            elif s[i] == '%' and i + 2 < len(s):
                hex_val = s[i+1:i+3]
                try:
                    # Attempt to convert the two characters after % to an integer
                    val = int(hex_val, 16)
                    res.append(chr(val))
                    i += 3
                except ValueError:
                    # If not valid hex, treat % as a literal character
                    res.append('%')
                    i += 1
            else:
                res.append(s[i])
                i += 1
        return "".join(res)

    pairs = []
    if not qs:
        return pairs

    # Split by '&' and iterate through components
    for part in qs.split('&'):
        if not part:
            continue
        
        if '=' in part:
            # Split only on the first occurrence of '='
            k, v = part.split('=', 1)
            pairs.append((unquote(k), unquote(v)))
        else:
            # Component with no '=' yields a pair with an empty value
            pairs.append((unquote(part), ""))
            
    return pairs
